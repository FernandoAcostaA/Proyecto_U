 3) CONTENIDO DE CADA TABLA
    (17 tablas de dominio + tablas internas de Laravel)
================================================================================

-- *****************************************************************************
-- 3.1 ROLES (admin, trabajador, cliente)
-- *****************************************************************************
SELECT * FROM roles ORDER BY id;

-- *****************************************************************************
-- 3.2 USUARIOS (administrador, trabajadores y clientes con acceso)
-- *****************************************************************************
SELECT u.id, u.email, u.rol_id, r.nombre AS rol, u.created_at
FROM usuarios u
JOIN roles r ON r.id = u.rol_id
ORDER BY u.id;

-- *****************************************************************************
-- 3.3 EMPLEADOS (peluqueros/barberos)
-- *****************************************************************************
SELECT * FROM empleados ORDER BY id;

-- *****************************************************************************
-- 3.4 SERVICIOS (cortes, tintes, barba, etc.)
-- *****************************************************************************
SELECT * FROM servicios ORDER BY id;

-- *****************************************************************************
-- 3.5 SERVICIO_EMPLEADO (relacion: que empleados dan cada servicio)
-- *****************************************************************************
SELECT * FROM servicio_empleado ORDER BY servicio_id, empleado_id;

-- *****************************************************************************
-- 3.6 SERVICIO_PRODUCTO (relacion: productos que consume cada servicio + cantidad)
-- *****************************************************************************
SELECT * FROM servicio_producto ORDER BY servicio_id;

-- *****************************************************************************
-- 3.7 PRODUCTOS (inventario)
-- *****************************************************************************
SELECT * FROM productos ORDER BY id;

-- *****************************************************************************
-- 3.8 PROVEEDORES
-- *****************************************************************************
SELECT * FROM proveedores ORDER BY id;

-- *****************************************************************************
-- 3.9 CLIENTES (usuario_id opcional cuando el cliente se registra)
-- *****************************************************************************
SELECT * FROM clientes ORDER BY id;

-- *****************************************************************************
-- 3.10 CITAS (agenda + verificacion de reservas publicas)
-- *****************************************************************************
SELECT * FROM citas ORDER BY id;

-- *****************************************************************************
-- 3.11 COMISIONES (comision generada por cita completada)
-- *****************************************************************************
SELECT * FROM comisiones ORDER BY id;

-- *****************************************************************************
-- 3.12 PAGOS
-- *****************************************************************************
SELECT * FROM pagos ORDER BY id;

-- *****************************************************************************
-- 3.13 HORARIOS_DISPONIBILIDAD (turnos de cada empleado por dia de semana 0-6)
-- *****************************************************************************
SELECT * FROM horarios_disponibilidad ORDER BY empleado_id, dia_semana;

-- *****************************************************************************
-- 3.14 MOVIMIENTOS_STOCK (entradas/salidas de inventario)
-- *****************************************************************************
SELECT * FROM movimientos_stock ORDER BY id;

-- *****************************************************************************
-- 3.15 RECORDATORIOS (notificaciones programadas de citas)
-- *****************************************************************************
SELECT * FROM recordatorios ORDER BY id;

-- *****************************************************************************
-- 3.16 TIPOS_CORTE (catalogo de estilos ofrecidos al cliente)
-- *****************************************************************************
SELECT * FROM tipos_corte ORDER BY id;

-- *****************************************************************************
-- 3.17 CALIFICACIONES (estrellas 1-5 sobre citas completadas)
-- *****************************************************************************
SELECT * FROM calificaciones ORDER BY id;
